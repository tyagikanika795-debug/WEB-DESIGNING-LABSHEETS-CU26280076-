```php
        header('Content-Type: application/json');

        exit;
    }

    {
        header("Location: $path");
        exit;
    }
```

## `core/Csrf.php`

```php
<?php
// Task 9: CSRF token generate / store / validate
class Csrf
{
    public static function token(): string
    {
        if (empty($_SESSION['csrf_token'])) {
            $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
        }
        return $_SESSION['csrf_token'];
    }

    public static function field(): string
    {
        return '<input type="hidden" name="csrf_token" value="' . self::token() . '">';
    }

    public static function verify(?string $token): bool
    {
        return !empty($_SESSION['csrf_token']) && is_string($token)
            && hash_equals($_SESSION['csrf_token'], $token);
    }

    public static function protect(): void
    {
        if (!self::verify($_POST['csrf_token'] ?? null)) {
            http_response_code(403);
            exit('Invalid CSRF token.');
        }
    }
}
```

```php
        $this->pdo = new PDO($dsn, $c['user'], $c['pass'], [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES => false,
        ]);
    }

    private function __clone() {}

    public static function getInstance(): PDO
    {
        if (self::$instance === null) {
            self::$instance = new Database();
        }
        return self::$instance->pdo;
    }
}
```

## `core/Model.php`

```php
<?php
// Task 5: reusable CRUD base model
abstract class Model
{
    protected string $table;
    protected PDO $db;

    public function __construct()
    {
        $this->db = Database::getInstance();
    }

    public function all(): array
    {
        return $this->db->query("SELECT * FROM {$this->table} ORDER BY id DESC")->fetchAll();
    }

    public function find(int $id): ?array
    {
        $st = $this->db->prepare("SELECT * FROM {$this->table} WHERE id = ?");
        $st->execute([$id]);
        return $st->fetch() ?: null;
    }

    public function create(array $data): int
    {
        $cols = implode(',', array_keys($data));
        $ph = implode(',', array_fill(0, count($data), '?'));
        $st = $this->db->prepare("INSERT INTO {$this->table} ($cols) VALUES ($ph)");
        $st->execute(array_values($data));
        return (int)$this->db->lastInsertId();
    }

    public function update(int $id, array $data): bool
    {
        $set = implode(',', array_map(fn($c) => "$c = ?", array_keys($data)));
        $st = $this->db->prepare("UPDATE {$this->table} SET $set WHERE id = ?");
        return $st->execute([...array_values($data), $id]);
    }

    public function delete(int $id): bool
    {
        $st = $this->db->prepare("DELETE FROM {$this->table} WHERE id = ?");
        return $st->execute([$id]);
    }
}
```

## `core/Router.php`

```php
<?php
// Task 4: simple router  (METHOD + path -> Controller@action)
class Router
{
    private array $routes = [];

    public function add(string $method, string $path, array $handler): void
    {
        $this->routes[] = [$method, $path, $handler];
    }

    public function dispatch(string $method, string $uri): void
    {
        $path = rtrim(parse_url($uri, PHP_URL_PATH), '/') ?: '/';
        foreach ($this->routes as [$m, $p, $handler]) {
            $regex = '#^' . preg_replace('#\{(\w+)\}#', '(?P<$1>[^/]+)', $p) . '$#';
            if ($m === $method && preg_match($regex, $path, $match)) {
                $params = array_filter($match, 'is_string', ARRAY_FILTER_USE_KEY);
                [$class, $action] = $handler;
                (new $class())->$action(...array_values($params));
                return;
            }
        }
        http_response_code(404);
        require __DIR__ . '/../views/errors/404.php';
    }
}
```

```php
class Session
{
    public static function start(): void
    {
        $https = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';
        ini_set('session.use_strict_mode', '1');
        ini_set('session.use_only_cookies', '1');
        session_set_cookie_params([
            'lifetime' => 0,
            'path' => '/',
            'secure' => $https,      // only over HTTPS
            'httponly' => true,      // not readable by JavaScript
            'samesite' => 'Strict',  // CSRF hardening
        ]);
        session_start();
    }

    public static function login(array $user): void
    {
        session_regenerate_id(true); // prevent session fixation
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['user_name'] = $user['name'];
    }

    public static function logout(): void
    {
        $_SESSION = [];
        session_destroy();
    }

    public static function check(): bool
    {
        return isset($_SESSION['user_id']);
    }
}
```

```php
class Validator
{
    public static function email(string $v): string|false
    {
        return filter_var(trim($v), FILTER_VALIDATE_EMAIL);
    }

    public static function url(string $v): string|false
    {
        return filter_var(trim($v), FILTER_VALIDATE_URL);
    }

    public static function int(mixed $v): int|false
    {
        return filter_var($v, FILTER_VALIDATE_INT);
    }

    public static function float(mixed $v): float|false
    {
        return filter_var($v, FILTER_VALIDATE_FLOAT);
    }

    // strip tags + special chars (sanitize strings)
    public static function string(string $v): string
    {
        return htmlspecialchars(strip_tags(trim($v)), ENT_QUOTES, 'UTF-8');
    }
}
```

## `controllers/ApiController.php`

```php
<?php
// Task 7 + 8: RESTful JSON API with token authentication
class ApiController extends Controller
{
    private Product $products;
    private User $users;

    public function __construct()
    {
        $this->products = new Product();
        $this->users = new User();
    }

    private function body(): array
    {
        return json_decode(file_get_contents('php://input'), true) ?? [];
    }

    private function authenticate(): void
    {
        $h = $_SERVER['HTTP_AUTHORIZATION'] ?? '';
        if (!preg_match('/^Bearer\s+(\S+)$/', $h, $m) || !$this->users->findByToken($m[1])) {
            $this->json(['error' => 'Unauthorized'], 401);
        }
    }

    // POST /api/login  -> returns token
    public function login(): void
    {
        $d = $this->body();
        $email = Validator::email($d['email'] ?? '');
        $user = $email ? $this->users->findByEmail($email) : null;
        if (!$user || $this->users->isLocked($user) || !password_verify($d['password'] ?? '', $user['password'])) {
            $this->json(['error' => 'Invalid credentials'], 401);
        }
        $this->json(['token' => $this->users->issueApiToken($user['id'])]);
    }

    public function index(): void
    {
        $this->authenticate();
        $this->json($this->products->all());
    }

    public function show(string $id): void
    {
        $this->authenticate();
        $p = $this->products->find((int)$id);
        $p ? $this->json($p) : $this->json(['error' => 'Not found'], 404);
    }

    private function clean(array $d): array|false
    {
        $name = Validator::string($d['name'] ?? '');
        $price = Validator::float($d['price'] ?? null);
        if ($name === '' || $price === false || $price < 0) {
            return false;
        }
        return ['name' => $name, 'price' => $price, 'description' => Validator::string($d['description'] ?? '')];
    }

    public function store(): void
    {
        $this->authenticate();
        $data = $this->clean($this->body());
        if (!$data) $this->json(['error' => 'Invalid input'], 422);
        $this->json(['id' => $this->products->create($data)], 201);
    }

    public function update(string $id): void
    {
        $this->authenticate();
        if (!$this->products->find((int)$id)) $this->json(['error' => 'Not found'], 404);
        $data = $this->clean($this->body());
        if (!$data) $this->json(['error' => 'Invalid input'], 422);
        $this->products->update((int)$id, $data);
        $this->json(['updated' => true]);
    }

    public function destroy(string $id): void
    {
        $this->authenticate();
        if (!$this->products->find((int)$id)) $this->json(['error' => 'Not found'], 404);
        $this->products->delete((int)$id);
        $this->json(['deleted' => true]);
    }
}
```


## `controllers/AuthController.php`

```php
<?php
class AuthController extends Controller
{
    private User $users;
    private array $cfg;

    public function __construct()
    {
        $this->users = new User();
        $this->cfg = require __DIR__ . '/../config/config.php';
    }

    // ---------- Register (Task 11 + 14) ----------
    public function registerForm(): void
    {
        $this->view('auth/register', ['title' => 'Register', 'error' => null]);
    }

    public function register(): void
    {
        Csrf::protect();
        $name = Validator::string($_POST['name'] ?? '');
        $email = Validator::email($_POST['email'] ?? '');
        $pass = $_POST['password'] ?? '';

        if ($name === '' || !$email || strlen($pass) < 8) {
            $this->view('auth/register', ['title' => 'Register', 'error' => 'Enter a name, valid email and 8+ char password.']);
            return;
        }
        if ($this->users->findByEmail($email)) {
            $this->view('auth/register', ['title' => 'Register', 'error' => 'Email already registered.']);
            return;
        }
        $this->users->create([
            'name' => $name,
            'email' => $email,
            'password' => password_hash($pass, PASSWORD_DEFAULT),
        ]);
        Mailer::send($email, 'Welcome!', "<h3>Hello $name</h3><p>Your registration is confirmed.</p>");
        $this->redirect('/login?registered=1');
    }

    // ---------- Login (Task 10 + 12) ----------
    public function loginForm(): void
    {
        $this->view('auth/login', ['title' => 'Login', 'error' => null]);
    }

    public function login(): void
    {
        Csrf::protect();
        $email = Validator::email($_POST['email'] ?? '');
        $pass = $_POST['password'] ?? '';
        $user = $email ? $this->users->findByEmail($email) : null;
        $error = 'Invalid email or password.';

        if ($user && $this->users->isLocked($user)) {
            $error = "Account locked. Try again after {$user['locked_until']}.";
        } elseif ($user && password_verify($pass, $user['password'])) {
            $this->users->clearFailures($user['id']);
            Session::login($user); // regenerates session id
            $this->redirect('/');
        } elseif ($user) {
            $this->users->registerFailure($user, $this->cfg['max_attempts'], $this->cfg['lock_minutes']);
        }
        $this->view('auth/login', ['title' => 'Login', 'error' => $error]);
    }

    public function logout(): void
    {
        Session::logout();
        $this->redirect('/login');
    }

    // ---------- Forgot password (Task 13) ----------
    public function forgotForm(): void
    {
        $this->view('auth/forgot', ['title' => 'Forgot Password', 'msg' => null]);
    }

    public function forgot(): void
    {
        Csrf::protect();
        $email = Validator::email($_POST['email'] ?? '');
        $user = $email ? $this->users->findByEmail($email) : null;
        if ($user) {
            $token = $this->users->createResetToken($user['id']);
            $link = $this->cfg['app_url'] . '/reset-password/' . $token;
            Mailer::send($email, 'Password reset', "<p>Click to reset (valid 1 hour): <a href='$link'>$link</a></p>");
        }
        // same message either way (does not reveal which emails exist)
        $this->view('auth/forgot', ['title' => 'Forgot Password', 'msg' => 'If that email exists, a reset link has been sent.']);
    }

    public function resetForm(string $token): void
    {
        $valid = $this->users->findValidReset($token) !== null;
        $this->view('auth/reset', ['title' => 'Reset Password', 'token' => $token, 'valid' => $valid, 'error' => null]);
    }

    public function reset(string $token): void
    {
        Csrf::protect();
        $reset = $this->users->findValidReset($token);
        $pass = $_POST['password'] ?? '';
        if (!$reset || strlen($pass) < 8) {
            $this->view('auth/reset', ['title' => 'Reset Password', 'token' => $token, 'valid' => (bool)$reset, 'error' => 'Invalid link or password too short.']);
            return;
        }
        $this->users->resetPassword($reset, $pass);
        $this->redirect('/login?reset=1');
    }
}
```

## `controllers/UploadController.php`

```php
<?php
// Task 15: secure file upload
class UploadController extends Controller
{
    public function form(): void
    {
        $this->view('pages/upload', ['title' => 'Upload', 'msg' => null]);
    }

    public function handle(): void
    {
        Csrf::protect();
        $cfg = (require __DIR__ . '/../config/config.php')['upload'];
        $f = $_FILES['file'] ?? null;
        $msg = null;

        if (!$f || $f['error'] !== UPLOAD_ERR_OK) {
            $msg = 'Upload failed.';
        } elseif ($f['size'] > $cfg['max_size']) {
            $msg = 'File too large (max 2 MB).';
        } else {
            $mime = (new finfo(FILEINFO_MIME_TYPE))->file($f['tmp_name']); // real MIME, not client-sent
            if (!isset($cfg['allowed'][$mime])) {
                $msg = 'File type not allowed.';
            } else {
                $newName = bin2hex(random_bytes(16)) . '.' . $cfg['allowed'][$mime]; // rename
                $msg = move_uploaded_file($f['tmp_name'], $cfg['dir'] . $newName)
                    ? "Uploaded as $newName" : 'Could not save file.';
            }
        }
        $this->view('pages/upload', ['title' => 'Upload', 'msg' => $msg]);
    }
}
```

```php
<?php
// Task 14: PHPMailer over SMTP  (composer require phpmailer/phpmailer)
use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

require_once __DIR__ . '/../vendor/autoload.php';

class Mailer
{
    public static function send(string $to, string $subject, string $html): bool
    {
        $c = (require __DIR__ . '/../config/config.php')['mail'];
        $mail = new PHPMailer(true);
        try {
            $mail->isSMTP();
            $mail->Host = $c['host'];
            $mail->SMTPAuth = true;
            $mail->Username = $c['user'];
            $mail->Password = $c['pass'];
            $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS;
            $mail->Port = $c['port'];
            $mail->setFrom($c['from'], $c['name']);
            $mail->addAddress($to);
            $mail->isHTML(true);
            $mail->Subject = $subject;
            $mail->Body = $html;
            $mail->send();
            return true;
        } catch (Exception $e) {
            error_log('Mailer error: ' . $mail->ErrorInfo);
            return false;
        }
    }
}
```

## `models/User.php`

```php
<?php
class User extends Model
{
    protected string $table = 'users';

    public function findByEmail(string $email): ?array
    {
        $st = $this->db->prepare('SELECT * FROM users WHERE email = ?');
        $st->execute([$email]);
        return $st->fetch() ?: null;
    }

    public function findByToken(string $token): ?array
    {
        $st = $this->db->prepare('SELECT * FROM users WHERE api_token = ?');
        $st->execute([hash('sha256', $token)]);
        return $st->fetch() ?: null;
    }

    public function issueApiToken(int $id): string
    {
        $token = bin2hex(random_bytes(32));
        $this->update($id, ['api_token' => hash('sha256', $token)]);
        return $token;
    }

    // Task 12: login attempt limiter
    public function registerFailure(array $user, int $max, int $minutes): void
    {
        $n = $user['failed_attempts'] + 1;
        if ($n >= $max) {
            $this->update($user['id'], [
                'failed_attempts' => 0,
                'locked_until' => date('Y-m-d H:i:s', time() + $minutes * 60),
            ]);
        } else {
            $this->update($user['id'], ['failed_attempts' => $n]);
        }
    }

    public function clearFailures(int $id): void
    {
        $this->update($id, ['failed_attempts' => 0, 'locked_until' => null]);
    }

    public function isLocked(array $user): bool
    {
        return !empty($user['locked_until']) && strtotime($user['locked_until']) > time();
    }

    // Task 13: forgot password
    public function createResetToken(int $userId): string
    {
        $token = bin2hex(random_bytes(32));
        $st = $this->db->prepare('INSERT INTO password_resets (user_id, token_hash, expires_at) VALUES (?,?,?)');
        $st->execute([$userId, hash('sha256', $token), date('Y-m-d H:i:s', time() + 3600)]);
        return $token;
    }

    public function findValidReset(string $token): ?array
    {
        $st = $this->db->prepare('SELECT * FROM password_resets WHERE token_hash = ? AND expires_at > NOW() AND used = 0');
        $st->execute([hash('sha256', $token)]);
        return $st->fetch() ?: null;
    }

    public function resetPassword(array $reset, string $newPassword): void
    {
        $this->update($reset['user_id'], ['password' => password_hash($newPassword, PASSWORD_DEFAULT)]);
        $this->db->prepare('UPDATE password_resets SET used = 1 WHERE id = ?')->execute([$reset['id']]);
    }
}
```

```apache
# Task 17: clean URLs, force HTTPS, no directory listing
Options -Indexes
DirectoryIndex index.php

RewriteEngine On

# Force HTTPS
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Pass Authorization header to PHP (needed for Bearer token)
RewriteCond %{HTTP:Authorization} ^(.*)
RewriteRule .* - [E=HTTP_AUTHORIZATION:%1]

# Clean URLs -> front controller
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^ index.php [L]

# Task 16: custom error pages for Apache-level errors
ErrorDocument 404 /index.php
```

## `public/js/app.js`

```javascript
// Task 8: consume the secured API with fetch()
const API = '/api/products';
let token = sessionStorage.getItem('api_token');
const $ = id => document.getElementById(id);
const status = msg => ($('status').textContent = msg);

function authHeaders() {
  return { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + token };
}

async function login() {
  const res = await fetch('/api/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: $('email').value, password: $('password').value })
  });
  const data = await res.json();
  if (!res.ok) return status(data.error || 'Login failed');
  token = data.token;
  sessionStorage.setItem('api_token', token);
  showCrud();
}

async function loadProducts() {
  const res = await fetch(API, { headers: authHeaders() });
  if (!res.ok) { sessionStorage.removeItem('api_token'); return status('Unauthorized - login again'); }
  const items = await res.json();
  $('list').innerHTML = items.map(p => `
    <tr>
      <td>${p.id}</td><td>${esc(p.name)}</td><td>${p.price}</td><td>${esc(p.description || '')}</td>
      <td><button onclick="editProduct(${p.id})">Edit</button>
          <button onclick="removeProduct(${p.id})">Delete</button></td>
    </tr>`).join('');
}

function esc(s) {
  const d = document.createElement('div'); d.textContent = s; return d.innerHTML;
}

async function addProduct() {
  const res = await fetch(API, {
    method: 'POST', headers: authHeaders(),
    body: JSON.stringify({ name: $('pname').value, price: $('pprice').value, description: $('pdesc').value })
  });
  status(res.ok ? 'Product added' : 'Invalid input');
  loadProducts();
}

async function editProduct(id) {
  const name = prompt('New name:'); const price = prompt('New price:');
  if (!name || !price) return;
  const res = await fetch(`${API}/${id}`, {
    method: 'PUT', headers: authHeaders(), body: JSON.stringify({ name, price })
  });
  status(res.ok ? 'Updated' : 'Update failed');
  loadProducts();
}

async function removeProduct(id) {
  if (!confirm('Delete this product?')) return;
  const res = await fetch(`${API}/${id}`, { method: 'DELETE', headers: authHeaders() });
  status(res.ok ? 'Deleted' : 'Delete failed');
  loadProducts();
}

function showCrud() {
  $('loginBox').hidden = true;
  $('crudBox').hidden = false;
  loadProducts();
}

$('loginBtn').addEventListener('click', login);
$('addBtn').addEventListener('click', addProduct);
if (token) showCrud();
```
