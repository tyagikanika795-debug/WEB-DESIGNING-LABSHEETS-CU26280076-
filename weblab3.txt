    <input type="number" id="num" placeholder="Enter a number">
    <button onclick="checkNumber()">Check</button>

    <div id="result"></div>

    <script>
        function checkNumber() {
            let n = Number(document.getElementById("num").value);
            let isPrime = true;

            if (n <= 1) {
                document.getElementById("result").innerHTML =
                    "The number is neither Prime nor Composite.";
                return;
            }

            for (let i = 2; i < n; i++) {
                if (n % i == 0) {
                    isPrime = false;
                    break;
                }
            }

            if (isPrime) {
                document.getElementById("result").innerHTML =
                    n + " is a Prime number.";
            } else {
                document.getElementById("result").innerHTML =
                    n + " is a Composite number.";
            }
        }
    </script>

    let n = Number(prompt("Enter number of terms:"));

    // Using for loop
    let a = 0, b = 1;
    let resultFor = "";

    for (let i = 1; i <= n; i++) {
        resultFor += a + " ";
        let c = a + b;
        a = b;
        b = c;
    }

    document.write("<h3>Using For Loop:</h3>");
    document.write(resultFor);


    
    a = 0;
    b = 1;
    let i = 1;
    let resultWhile = "";

    while (i <= n) {
        resultWhile += a + " ";
        let c = a + b;
        a = b;
        b = c;
        i++;
    }

    document.write("<h3>Using While Loop:</h3>");
    document.write(resultWhile);


    
    a = 0;
    b = 1;
    i = 1;
    let resultDoWhile = "";

    if (n > 0) {
        do {
            resultDoWhile += a + " ";
            let c = a + b;
            a = b;
            b = c;
            i++;
        } while (i <= n);
    }

    document.write("<h3>Using Do-While Loop:</h3>");
    document.write(resultDoWhile);
</script>

            let ch = text[i].toLowerCase();

                ch == "o" || ch == "u") {

        }

</head>
<body>

<h2>Dynamic Table Generator</h2>

<label>Enter Rows:</label>
<input type="number" id="rows">

<br><br>

<label>Enter Columns:</label>
<input type="number" id="columns">

<br><br>

<button onclick="createTable()">Generate Table</button>

<div id="tableContainer"></div>

<script>
    function createTable() {

        let rows = Number(document.getElementById("rows").value);
        let columns = Number(document.getElementById("columns").value);

        let table = document.createElement("table");
        table.border = "1";

        for (let i = 1; i <= rows; i++) {

            let row = document.createElement("tr");

            for (let j = 1; j <= columns; j++) {

                let cell = document.createElement("td");

                cell.textContent = "Row " + i + ", Column " + j;

                row.appendChild(cell);
            }

            table.appendChild(row);
        }

        let container = document.getElementById("tableContainer");

        container.innerHTML = "";
        container.appendChild(table);
    }
</script>

</body>
